// Dominio oficial en GitHub Pages para la visualización digital del PDF const BASE_VIEWER_URL = "https://amarintrans.github.io/visor_carta_porte.html"; async function processPDF(action) { try { if (!window.jspdf || !window.QRCode) { alert("Las librerías requeridas no se han cargado aún. Comprueba la conexión."); return; } const { jsPDF } = window.jspdf; const doc = new jsPDF(); const orderId = "TR-" + Math.floor(100000 + Math.random() * 900000); const carrier = document.getElementById('carrierName').value; const nif = document.getElementById('carrierNif').value; const plate = document.getElementById('plate').value; const client = document.getElementById('clientName').value; const clientCif = document.getElementById('clientCif').value; const origin = document.getElementById('origin').value; const destName = document.getElementById('destName').value; const destCif = document.getElementById('destCif').value; const destination = document.getElementById('destination').value; const cargo = document.getElementById('cargo').value; const weight = document.getElementById('weight').value; const packages = document.getElementById('packages').value; const base = parseFloat(document.getElementById('basePrice').value) || 0; const vatRate = parseFloat(document.getElementById('vatRate').value) || 0; const irpfRate = parseFloat(document.getElementById('irpfRate').value) || 0; const vat = base * (vatRate / 100); const irpf = base * (irpfRate / 100); const total = base + vat - irpf; // Construcción de la URL hospedada en amarintrans.github.io const queryParams = new URLSearchParams({ ref: orderId, carrier: carrier, nif: nif, plate: plate, client: client, clientCif: clientCif, origin: origin, destName: destName, destCif: destCif, destination: destination, cargo: cargo, weight: weight, packages: packages, total: total.toFixed(2) + "€" }); const qrViewerURL = `${BASE_VIEWER_URL}?${queryParams.toString()}`; // Generar la imagen del QR con el enlace a GitHub Pages const qrImageBase64 = await buildQRCodeDataURL(qrViewerURL); // Encabezado PDF doc.setFillColor(30, 58, 138); doc.rect(0, 0, 210, 28, 'F'); doc.setTextColor(255, 255, 255); doc.setFontSize(16); doc.text("ORDEN DE TRANSPORTE / CARTA DE PORTE", 14, 18); doc.setFontSize(10); doc.text(`Nº Ref: ${orderId}`, 160, 18); doc.setTextColor(15, 23, 42); // 1. Transportista doc.setFont("helvetica", "bold"); doc.setFontSize(10); doc.text("1. TRANSPORTISTA (AUTÓNOMO)", 14, 38); doc.setFont("helvetica", "normal"); doc.text(`Nombre: ${carrier}`, 14, 44); doc.text(`NIF/CIF: ${nif}`, 14, 50); doc.text(`Matrícula: ${plate}`, 14, 56); // 2. Cargador doc.setFont("helvetica", "bold"); doc.text("2. CARGADOR (CLIENTE)", 110, 38); doc.setFont("helvetica", "normal"); doc.text(`Cliente: ${client}`, 110, 44); doc.text(`CIF/NIF: ${clientCif}`, 110, 50); doc.text(`Origen: ${origin}`, 110, 56); doc.setDrawColor(203, 213, 225); doc.line(14, 62, 196, 62); // 3. Destinatario doc.setFont("helvetica", "bold"); doc.text("3. DESTINATARIO", 14, 70); doc.setFont("helvetica", "normal"); doc.text(`Destinatario: ${destName}`, 14, 76); doc.text(`CIF/NIF: ${destCif}`, 14, 82); doc.text(`Destino: ${destination}`, 14, 88); // 4. Mercancía doc.setFont("helvetica", "bold"); doc.text("4. DETALLE DE MERCANCÍA", 110, 70); doc.setFont("helvetica", "normal"); doc.text(`Carga: ${cargo}`, 110, 76); doc.text(`Peso: ${weight} kg`, 110, 82); doc.text(`Bultos: ${packages}`, 110, 88); doc.line(14, 95, 196, 95); // 5. Liquidación Económica doc.setFont("helvetica", "bold"); doc.text("5. LIQUIDACIÓN ECONÓMICA", 14, 105); doc.setFont("helvetica", "normal"); doc.text(`Base Imponible: ${base.toFixed(2)} €`, 14, 113); doc.text(`IVA (${vatRate}%): ${vat.toFixed(2)} €`, 14, 119); doc.text(`Retención IRPF (-${irpfRate}%): -${irpf.toFixed(2)} €`, 14, 125); doc.setFont("helvetica", "bold"); doc.setFontSize(11); doc.text(`TOTAL A PERCIBIR: ${total.toFixed(2)} €`, 14, 135); // Estampar Código QR de consulta en línea doc.addImage(qrImageBase64, 'PNG', 150, 102, 40, 40); doc.setFontSize(8); doc.setFont("helvetica", "normal"); doc.text("Escanea para ver online", 150, 145); // Firmas doc.line(14, 165, 196, 165); doc.setFontSize(9); doc.setFont("helvetica", "bold"); doc.text("Firma Transportista:", 20, 173); doc.text("Sello / Firma Destinatario:", 120, 173); const fileName = `Carta_Porte_${orderId}.pdf`; if (action === 'download') { doc.save(fileName); } else if (action === 'share') { const pdfBlob = doc.output('blob'); const pdfFile = new File([pdfBlob], fileName, { type: 'application/pdf' }); if (navigator.canShare && navigator.canShare({ files: [pdfFile] })) { await navigator.share({ title: `Carta de Porte ${orderId}`, text: `Carta de Porte Ref. ${orderId}. Ver digitalmente: ${qrViewerURL}`, files: [pdfFile] }); } else { doc.save(fileName); const bodyMsg = encodeURIComponent(`Hola,\n\nTe adjunto la Carta de Porte Ref: ${orderId}.\nPuedes consultar el documento online en amarintrans:\n${qrViewerURL}`); window.location.href = `mailto:?subject=Carta de Porte ${orderId}&body=${bodyMsg}`; } } } catch (error) { console.error("Error al procesar el PDF:", error); alert("No se pudo procesar o compartir el archivo PDF."); } }
